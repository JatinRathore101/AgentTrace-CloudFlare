Schema

-- src/agent/schema.ts — executed in TraceAgent.onStart() with this.sql; idempotent
CREATE TABLE IF NOT EXISTS runs (
  id TEXT PRIMARY KEY,               -- crypto.randomUUID()
  scenario TEXT NOT NULL,            -- 'lisbon-weekend' | 'tokyo-weather' | 'budget-check'
  goal TEXT NOT NULL,
  faults TEXT NOT NULL DEFAULT '[]', -- JSON array of fault flags
  model TEXT NOT NULL,
  status TEXT NOT NULL,              -- running | succeeded | failed | max_steps
  final_answer TEXT,
  error TEXT,
  steps INTEGER NOT NULL DEFAULT 0,
  prompt_tokens INTEGER NOT NULL DEFAULT 0,
  completion_tokens INTEGER NOT NULL DEFAULT 0,
  neurons REAL NOT NULL DEFAULT 0,
  cost_usd REAL NOT NULL DEFAULT 0,
  started_at INTEGER NOT NULL,       -- epoch ms
  ended_at INTEGER
);
CREATE TABLE IF NOT EXISTS spans (
  id TEXT PRIMARY KEY,
  run_id TEXT NOT NULL REFERENCES runs(id),
  seq INTEGER NOT NULL,              -- 1..n in execution order
  step INTEGER NOT NULL,             -- agent loop iteration
  kind TEXT NOT NULL,                -- llm | tool
  name TEXT NOT NULL,                -- model id or tool name
  status TEXT NOT NULL,              -- ok | error
  input TEXT,                        -- JSON, truncated to 32 KB
  output TEXT,                       -- JSON, truncated to 32 KB
  error TEXT,
  prompt_tokens INTEGER,
  completion_tokens INTEGER,
  started_at INTEGER NOT NULL,
  ended_at INTEGER NOT NULL,
  latency_ms INTEGER NOT NULL,
  gateway_log_id TEXT
);
CREATE INDEX IF NOT EXISTS idx_spans_run ON spans(run_id, seq);
CREATE TABLE IF NOT EXISTS findings (
  id TEXT PRIMARY KEY,
  run_id TEXT NOT NULL REFERENCES runs(id),
  detector TEXT NOT NULL,
  severity TEXT NOT NULL,            -- info | warn | crit
  span_seq INTEGER,
  message TEXT NOT NULL,
  evidence TEXT                      -- JSON
);
CREATE INDEX IF NOT EXISTS idx_findings_run ON findings(run_id);
CREATE TABLE IF NOT EXISTS chat_messages (
  id TEXT PRIMARY KEY,
  run_id TEXT NOT NULL REFERENCES runs(id),
  role TEXT NOT NULL,                -- user | assistant
  content TEXT NOT NULL,
  prompt_tokens INTEGER,
  completion_tokens INTEGER,
  created_at INTEGER NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_chat_run ON chat_messages(run_id, created_at);
📌
No external migrations

CREATE ... IF NOT EXISTS runs on every DO start, so it is idempotent. A future column change is done with ALTER TABLE guarded by a PRAGMA table_info check. Free plan limits: 2 MB per row, 100 bound parameters per query.

The implemented schema adds runs.is_demo (seeded rows, excluded from the daily cap), spans.neurons and chat_messages.neurons (Neurons reported by Workers AI), and an index on runs(started_at).

Demo seed

  • Fixture: src/agent/fixtures/demo-runs.json, containing 5 runs captured after calibration (todo section 4), each with run, spans[] and findings[].
  • Inserted only when runs is empty and the var SEED_DEMO = "true". It never overwrites real runs, so it is idempotent (TC25, TC26).
  • Seeded runs keep their original ids and timestamps. The scenario name is unchanged, and the runs list shows a demo chip.
// TraceAgent.onStart(): after the schema
if (this.env.SEED_DEMO === "true") {
  const [{ n }] = this.sql`SELECT count(*) AS n FROM runs`;
  if (n === 0) {
    for (const r of demoRuns) {            // import demoRuns from "./fixtures/demo-runs.json"
      insertRun(r.run); r.spans.forEach(insertSpan); r.findings.forEach(insertFinding);
    }
  }
}
